# clean venv from this file and running the suite; see QA.md §2A.
Django>=5.2,<5.3
djangorestframework>=3.16,<3.18
django-cors-headers>=4.4,<5.0

# materials/validators.py — real content-type sniffing (needs the system libmagic library, already
# present on a typical Linux install; confirmed present in this project's own dev environment) and
# an optional ClamAV scan client (talks to an already-running clamd daemon — installing ClamAV
# itself is a separate, real system package this pip requirement does not provide).
python-magic>=0.4.27,<0.5
clamd>=1.0.2,<2.0
# services/messages.py — user-to-user messaging, the industry-standard, actively-maintained Django
# app for this (verified against PyPI directly before choosing it: latest release 2026-03, vs. the
# older django-messages alternative's last release in 2019). django.contrib.sites is a genuine,
# confirmed-necessary dependency of django-postman itself (its own api.py unconditionally imports
# the Site model at load time), not an optional extra — see config/settings.py's own note on this.
django-postman>=4.6,<5.0
# config/sanitize.py — real server-side HTML sanitization for user-submitted exercise/material
# content (CLAUDE.md Section 11's own "sanitize on save too, defense in depth" requirement — never
# actually implemented until a security scan of the whole project found the gap). The frontend's own
# DOMPurify pass (renderContent.ts) was, until this fix, the ONLY sanitization layer for content
# that publishes with no moderator review at all (a verified contributor's new exercises).
bleach>=6.1,<7.0

redis>=5.0,<9.0
